-- ============================================================
-- DATABASE SCHEMA
-- ============================================================

-- ============================================================
-- USER
-- ============================================================
CREATE TABLE app_user (
    user_id        NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    name           VARCHAR2(60 CHAR) NOT NULL,
    password_hash  VARCHAR2(255 CHAR) NOT NULL,
    email          VARCHAR2(254 CHAR) NOT NULL,
    mobile_no      NUMBER(10) NOT NULL,
    dob            DATE NOT NULL,
    status         VARCHAR2(20 CHAR) DEFAULT 'ACTIVE' NOT NULL,
    created_at     TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at     TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_app_user PRIMARY KEY (user_id),
    CONSTRAINT uq_app_user_email UNIQUE (email),
    CONSTRAINT ck_app_user_status
        CHECK (status IN ('ACTIVE', 'INACTIVE', 'DELETED'))
);

CREATE INDEX ix_app_user_mobile ON app_user (mobile_no);

CREATE OR REPLACE TRIGGER trg_app_user_bu
BEFORE UPDATE ON app_user
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSTIMESTAMP;
END;
/

-- ============================================================
-- CONTACT
-- ============================================================
CREATE TABLE contact (
    contact_id      NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    owner_user_id   NUMBER NOT NULL,
    linked_user_id  NUMBER,
    contact_name    VARCHAR2(20 CHAR) NOT NULL,
    contact_phone   NUMBER(10) NOT NULL,
    created_at      TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at      TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_contact PRIMARY KEY (contact_id),
    CONSTRAINT ck_contact_not_self
        CHECK (
            linked_user_id IS NULL
            OR linked_user_id <> owner_user_id
        )
);

CREATE INDEX ix_contact_owner ON contact (owner_user_id);
CREATE INDEX ix_contact_linked ON contact (linked_user_id);

CREATE OR REPLACE TRIGGER trg_contact_bu
BEFORE UPDATE ON contact
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSTIMESTAMP;
END;
/

-- ============================================================
-- ACCOUNT
-- ============================================================
CREATE TABLE account (
    account_id  NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    user_id     NUMBER NOT NULL,
    balance     NUMBER(20,2) DEFAULT 0 NOT NULL,
    status      VARCHAR2(20 CHAR) DEFAULT 'ACTIVE' NOT NULL,
    created_at  TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at  TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_account PRIMARY KEY (account_id),
    CONSTRAINT ck_account_balance CHECK (balance >= 0),
    CONSTRAINT ck_account_status
        CHECK (status IN ('ACTIVE', 'FROZEN', 'CLOSED'))
);

CREATE INDEX ix_account_user ON account (user_id);

CREATE OR REPLACE TRIGGER trg_account_bu
BEFORE UPDATE ON account
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSTIMESTAMP;
END;
/

-- ============================================================
-- WALLET
-- ============================================================
CREATE TABLE wallet (
    wallet_id   NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    user_id     NUMBER NOT NULL,
    balance     NUMBER(20,2) DEFAULT 0 NOT NULL,
    created_at  TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
    updated_at  TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,

    CONSTRAINT pk_wallet PRIMARY KEY (wallet_id),
    CONSTRAINT uq_wallet_user UNIQUE (user_id),
    CONSTRAINT ck_wallet_balance CHECK (balance >= 0)
);

CREATE OR REPLACE TRIGGER trg_wallet_bu
BEFORE UPDATE ON wallet
FOR EACH ROW
BEGIN
    :NEW.updated_at := SYSTIMESTAMP;
END;
/

-- ============================================================
-- TRANSACTIONS
-- ============================================================
CREATE TABLE transactions (
    transaction_id        NUMBER GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    transaction_type      VARCHAR2(3 CHAR) NOT NULL,
    source_wallet_id      NUMBER,
    source_account_id     NUMBER,
    destination_wallet_id NUMBER NOT NULL,
    amount                NUMBER(20,2) NOT NULL,
    status                VARCHAR2(20 CHAR) DEFAULT 'PENDING' NOT NULL,
    created_at            TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
    completed_at          TIMESTAMP(6),

    CONSTRAINT pk_app_transaction PRIMARY KEY (transaction_id),

    CONSTRAINT fk_tx_source_wallet
        FOREIGN KEY (source_wallet_id)
        REFERENCES wallet (wallet_id),

    CONSTRAINT fk_tx_source_account
        FOREIGN KEY (source_account_id)
        REFERENCES account (account_id),

    CONSTRAINT fk_tx_destination_wallet
        FOREIGN KEY (destination_wallet_id)
        REFERENCES wallet (wallet_id),

    CONSTRAINT ck_tx_amount CHECK (amount > 0),

    CONSTRAINT ck_tx_type
        CHECK (transaction_type IN ('A2W', 'W2W')),

    CONSTRAINT ck_tx_status
        CHECK (status IN ('PENDING', 'COMPLETED', 'FAILED')),

    CONSTRAINT ck_tx_source_by_type
        CHECK (
            (transaction_type = 'W2W'
                AND source_wallet_id IS NOT NULL
                AND source_account_id IS NULL)
            OR
            (transaction_type = 'A2W'
                AND source_wallet_id IS NULL
                AND source_account_id IS NOT NULL)
        ),

    CONSTRAINT ck_tx_different_wallets
        CHECK (
            source_wallet_id IS NULL
            OR source_wallet_id <> destination_wallet_id
        ),

    CONSTRAINT ck_tx_completion_time
        CHECK (
            (status IN ('PENDING', 'FAILED') AND completed_at IS NULL)
            OR
            (status = 'COMPLETED' AND completed_at IS NOT NULL)
        )
);

CREATE INDEX ix_tx_source_wallet
    ON transactions (source_wallet_id);

CREATE INDEX ix_tx_source_account
    ON transactions (source_account_id);

CREATE INDEX ix_tx_destination_wallet
    ON transactions (destination_wallet_id);

CREATE INDEX ix_tx_created_at
    ON transactions (created_at);

-- ============================================================
-- OUTBOX EVENT
-- ============================================================
CREATE TABLE outbox_event (
    event_id        VARCHAR2(36 CHAR) NOT NULL,
    event_type      VARCHAR2(100 CHAR) NOT NULL,
    aggregate_id    NUMBER(19) NOT NULL,
    payload         CLOB NOT NULL,
    created_at      TIMESTAMP(6) NOT NULL,
    published_at    TIMESTAMP(6) NULL,
    publish_attempts NUMBER(10) DEFAULT 0 NOT NULL,
    last_error      VARCHAR2(1000 CHAR) NULL,

    CONSTRAINT pk_outbox_event PRIMARY KEY (event_id),
    CONSTRAINT ck_outbox_event_attempts
        CHECK (publish_attempts >= 0)
);

CREATE INDEX idx_outbox_event_pending
    ON outbox_event (published_at, created_at);
